문제

제네릭 클래스 ArrayList 및 Collections를 바탕으로 점수 및 단어를 정렬하는 문제입니다 이를 해결하는 다음 프로그램에 대해 빈칸을 채우세요 
86 89 90
apple banana carrot

코드 빈칸 채우기

import java.util.*;  
	  
public class GenericTest {  
	public static void main( String[] args ) { 
		<> scores =  <>();

		( int i = 0; i < scores.(); i )  
		    System.out.print( scores.(i) + " "  );  
		System.out.println();  
	  
		<> words =  <>();  
		words.( "carrot" );

		( int i = 0; i < words.(); i )  
		    System.out.print( words.(i) + " "  );  
	  
	}  
}
